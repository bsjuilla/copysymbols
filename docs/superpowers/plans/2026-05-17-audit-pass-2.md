```ts
"use client";

import { useCallback } from "react";

/**
 * Copies text to clipboard and shows the global toast.
 * Returns a stable callback ready to attach to onClick.
 */
export function useCopyToast() {
  return useCallback(async (text: string, label?: string) => {
    try {
      await navigator.clipboard.writeText(text);
    } catch {
      const ta = document.createElement("textarea");
      ta.value = text;
      ta.style.position = "fixed";
      ta.style.top = "-9999px";
      document.body.appendChild(ta);
      ta.select();
      try { document.execCommand("copy"); } catch {}
      document.body.removeChild(ta);
    }
    const toast = document.getElementById("global-toast");
    if (!toast) return;
    const sym = document.getElementById("toast-symbol");
    const msg = document.getElementById("toast-message");
    if (sym) sym.textContent = text.length > 20 ? text.slice(0, 20) + "…" : text;
    if (msg) msg.textContent = label ? `Copied ${label}!` : "Copied!";
    toast.classList.add("show");
    setTimeout(() => toast.classList.remove("show"), 1500);
  }, []);
}
```

```ts
import type { Translator } from "./index";

const VOWELS = new Set(["a", "e", "i", "o", "u", "A", "E", "I", "O", "U"]);

function encodeWord(word: string): string {
  if (!word) return word;
  const isUpper = word[0] === word[0].toUpperCase() && word[0] !== word[0].toLowerCase();
  const lower = word.toLowerCase();
  if (VOWELS.has(lower[0])) {
    const result = lower + "way";
    return isUpper ? result[0].toUpperCase() + result.slice(1) : result;
  }
  // find first vowel
  let split = 0;
  for (let i = 0; i < lower.length; i++) {
    if (VOWELS.has(lower[i])) { split = i; break; }
    if (i === lower.length - 1) { split = lower.length; }
  }
  const result = lower.slice(split) + lower.slice(0, split) + "ay";
  return isUpper ? result[0].toUpperCase() + result.slice(1) : result;
}

function encode(input: string): string {
  return input.split(/(\s+|[^\w'\s]+)/).map(token => {
    if (/^\s+$/.test(token) || /^[^\w'\s]+$/.test(token)) return token;
    return encodeWord(token);
  }).join("");
}
```
